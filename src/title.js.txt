    if (n === 'say') return A.say(el);
    const sc = tEl.querySelector('#tovl .mbody'), top = sc ? sc.scrollTop : 0;
    if (n === 'setdiff') { S.diff = el.dataset.v; save(); } else A[n](el, e);

      if (a && T_ACTIONS.includes(a.dataset.a)) {
        this.act(a, e);
        return;
      }
      const t = e.target.closest('[data-t]');
      if (t) {
        const k = t.dataset.t;
        if (k === 'play') this.play();
        else if (k === 'modal') this.open(t.dataset.p);

        return;
      }
      if (e.target.id === 'tovl') return this.close();
    });
    document.addEventListener('keydown', e => {
      if (!this.on) return;
      if (e.key === 'Escape' && this.modal) { e.preventDefault(); this.close(); return; }
      if ((e.key === 'Enter' || e.key === ' ') && !this.modal && !(e.target.closest && e.target.closest('button'))) { e.preventDefault(); this.play(); }
    });
    if (location.hash === '#play') this.play(); else this.show();
  }

/* ---------- menu jeda di dalam permainan: lanjut, kembali ke peta, atau ke menu depan ---------- */
const PM = {
  el: null, ctx: null,
  get on() { return !!(this.el && this.el.classList.contains('show')); },
  open(ctx) {
    this.ctx = ctx;
    if (!this.el) {
      this.el = document.createElement('div'); this.el.id = 'gmenu'; document.body.appendChild(this.el);
      this.el.addEventListener('click', e => { const b = e.target.closest('[data-g]'); if (b) this.pick(b.dataset.g); else if (e.target === this.el) this.close(); });
      document.addEventListener('keydown', e => { if (this.on && e.key === 'Escape') { e.preventDefault(); this.close(); } });
    }
    this.el.innerHTML = `<div class="mpanel" role="dialog" aria-label="Menu"><div class="mhead"><h3>Menu</h3><button class="mclose" data-g="resume" aria-label="Tutup">✕</button></div><div class="mbody">
      <div class="pmlist"><button class="btn" data-g="resume">▶ Lanjutkan</button><button class="btn ghost" data-g="map">${forestIcon('map', 'fui-inline')} ${ctx.mapLabel || 'Kembali ke peta'}</button><button class="btn ghost" data-g="title">${commonIcon('home', 'cui-inline')} Menu depan</button></div>
      ${ctx.note ? `<p class="note">${ctx.note}</p>` : ''}</div></div>`;
    this.el.classList.add('show');
    const b = this.el.querySelector('.pmlist .btn'); if (b) b.focus({ preventScroll: true });
  },
  close() { if (this.el) { this.el.classList.remove('show'); this.el.innerHTML = ''; } },
  pick(k) { const c = this.ctx; this.close(); if (k === 'map') c.toMap(); else if (k === 'title') c.toTitle(); }
};
